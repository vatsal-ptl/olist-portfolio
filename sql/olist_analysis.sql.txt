-- Q1: Total Revenue and Orders by Status

SELECT o.order_status, COUNT(DISTINCT o.order_id) AS total_orders,
    ROUND(SUM(oi.price), 2)    AS total_revenue
FROM olist_orders_dataset o
JOIN olist_order_items_dataset oi
    ON o.order_id = oi.order_id
GROUP BY o.order_status
ORDER BY total_revenue DESC;

-- Q2: Top 10 Categories by Revenue

SELECT t.product_category_name_english, ROUND(SUM(oi.price), 2) AS total_revenue
FROM olist_order_items_dataset oi
JOIN olist_products_dataset p
    ON oi.product_id = p.product_id
JOIN product_category_name_translation t
    ON p.product_category_name = t.product_category_name
GROUP BY t.product_category_name_english
ORDER BY total_revenue DESC
LIMIT 10;

-- Q3: Customers distribution across Brazil, and the percentage of total customers each state represents 

WITH by_state AS (
    SELECT
        customer_state,
        COUNT(customer_unique_id) AS total_by_state
    FROM olist_customers_dataset
    GROUP BY customer_state
)
SELECT
    customer_state,
    total_by_state,
    ROUND(total_by_state / SUM(total_by_state) OVER() * 100, 2) AS pct_of_total
FROM by_state
ORDER BY total_by_state DESC;

-- Q4: Top 10 Sellers by Revenue

SELECT oi.seller_id, s.seller_city, s.seller_state,
    ROUND(SUM(oi.price), 2) AS total_revenue
FROM olist_order_items_dataset oi
JOIN olist_sellers_dataset s
    ON oi.seller_id = s.seller_id
GROUP BY oi.seller_id, s.seller_city, s.seller_state
ORDER BY total_revenue DESC
LIMIT 10;

-- Q5: Underperforming Categories by Avg Payment Value

WITH cte AS (
    SELECT
        pt.product_category_name_english AS category_name,
        AVG(op.payment_value)            AS avg_payment
    FROM olist_order_items_dataset oi
    JOIN olist_order_payments_dataset op ON oi.order_id = op.order_id
    JOIN olist_products_dataset p        ON oi.product_id = p.product_id
    JOIN product_category_name_translation pt
        ON pt.product_category_name = p.product_category_name
    GROUP BY pt.product_category_name_english
),
overall AS (
    SELECT AVG(avg_payment) AS overall_avg FROM cte
)

SELECT cte.category_name,
    ROUND(cte.avg_payment, 2)      AS avg_payment,
    ROUND(overall.overall_avg, 2)  AS overall_avg
FROM cte CROSS JOIN overall
WHERE cte.avg_payment < overall.overall_avg
ORDER BY cte.avg_payment ASC
LIMIT 10;

-- Q6: Most Popular Payment Methods

SELECT payment_type, COUNT(payment_type) AS total_transactions
FROM olist_order_payments_dataset
GROUP BY payment_type
ORDER BY total_transactions DESC;

-- Q7: The revenue rank and percentage share of total revenue for every product category

WITH cat_rev AS (
    SELECT
        pt.product_category_name_english AS cat_in_eng,
        SUM(oi.price)                    AS cat_revenue
    FROM olist_products_dataset p
    JOIN olist_order_items_dataset oi ON p.product_id = oi.product_id
    JOIN product_category_name_translation pt
        ON p.product_category_name = pt.product_category_name
    GROUP BY cat_in_eng
)

SELECT RANK() OVER(ORDER BY cat_revenue DESC)          AS revenue_rank,
    cat_in_eng,
    ROUND(cat_revenue, 2)                           AS cat_revenue,
    ROUND(cat_revenue / SUM(cat_revenue) OVER()     AS pct_of_total
          * 100, 2)
FROM cat_rev
ORDER BY revenue_rank;

-- Q8: Month-over-Month Revenue Change

WITH monthly_rev AS (
    SELECT
        DATE_FORMAT(o.order_purchase_timestamp, '%Y-%m') AS month,
        ROUND(SUM(oi.price + oi.freight_value), 2)       AS revenue
    FROM olist_orders_dataset o
    JOIN olist_order_items_dataset oi ON o.order_id = oi.order_id
    GROUP BY month
)
SELECT month, revenue,
    LAG(revenue) OVER(ORDER BY month)                           AS prev_month_rev,
    ROUND(revenue - LAG(revenue) OVER(ORDER BY month), 2)      AS mom_change
FROM monthly_rev
ORDER BY month;

-- Q9: Top 2 Sellers per Category

WITH seller_cat_rev AS (
    SELECT pt.product_category_name_english AS category,
        oi.seller_id,
        ROUND(SUM(oi.price), 2) AS revenue,
        RANK() OVER(
            PARTITION BY pt.product_category_name_english
            ORDER BY SUM(oi.price) DESC
        ) AS rnk
    FROM olist_order_items_dataset oi
    JOIN olist_products_dataset p ON p.product_id = oi.product_id
    JOIN product_category_name_translation pt
        ON pt.product_category_name = p.product_category_name
    GROUP BY pt.product_category_name_english, oi.seller_id
)
SELECT *
FROM seller_cat_rev
WHERE rnk <= 2
ORDER BY category, rnk;

-- Q10: Customer Retention (2017 to 2018)

WITH order_in_2017 AS (
    SELECT c.customer_unique_id,
           COUNT(o.customer_id) AS ttl_odr_2017
    FROM olist_orders_dataset o
    JOIN olist_customers_dataset c ON o.customer_id = c.customer_id
    WHERE o.order_status = 'delivered'
      AND YEAR(o.order_purchase_timestamp) = 2017
    GROUP BY c.customer_unique_id
),
order_in_2018 AS (
    SELECT c.customer_unique_id,
           COUNT(o.customer_id) AS ttl_odr_2018
    FROM olist_orders_dataset o
    JOIN olist_customers_dataset c ON o.customer_id = c.customer_id
    WHERE o.order_status = 'delivered'
      AND YEAR(o.order_purchase_timestamp) = 2018
    GROUP BY c.customer_unique_id
)
SELECT
    order_in_2017.customer_unique_id,
    ttl_odr_2017,
    ttl_odr_2018,
    ttl_odr_2017 + ttl_odr_2018 AS total_orders
FROM order_in_2017
JOIN order_in_2018
    ON order_in_2017.customer_unique_id = order_in_2018.customer_unique_id
ORDER BY total_orders DESC;
 
-- Headline count:
SELECT COUNT(*) AS retained_customers
FROM order_in_2017
JOIN order_in_2018
    ON order_in_2017.customer_unique_id = order_in_2018.customer_unique_id;
